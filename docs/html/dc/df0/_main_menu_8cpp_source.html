482void MainMenu::setupComboboxSources() {
483 static std::unordered_map<unsigned char, unsigned char> shapeOrder = {
484 {1, 1}, {2, 4}, {3, 2}, {4, 3}, {5, 5}, {6, 0}};
485 std::function<bool(const Aperture &, const Aperture &)> apertureComparator =
486 [](const Aperture &a, const Aperture &b) {
487 if (a.apertureShapeID != b.apertureShapeID) {
488 return shapeOrder[a.apertureShapeID] < shapeOrder[b.apertureShapeID];
489 }
490 return a.width < b.width;
491 };
492
493 std::function<bool(const SideIron &, const SideIron &)> sideIronComparator =
494 [](const SideIron &a, const SideIron &b) { return a.length < b.length; };
495
496 DrawingComponentManager<Product>::addCallback(
497 [this]() { productSource.updateSource(); });
498 DrawingComponentManager<Aperture>::addCallback(
499 [this]() { apertureSource.updateSource(); });
500 DrawingComponentManager<ApertureShape>::addCallback(
501 [this]() { apertureShapeSource.updateSource(); });
502 DrawingComponentManager<Material>::addCallback([this]() {
503 topMaterialSource.updateSource();
504 bottomMaterialSource.updateSource();
505 });
506 DrawingComponentManager<SideIron>::addCallback([this, sideIronComparator]() {
507 sideIronSource.updateSource();
508 sideIronSource.sort(sideIronComparator);
509 });
510 DrawingComponentManager<Machine>::addCallback([this]() {
511 machineManufacturerSource.updateSource();
512 machineManufacturerSource.makeDistinct();
513 machineModelSource.updateSource();
514 });
515 DrawingComponentManager<MachineDeck>::addCallback(
516 [this]() { machineDeckSource.updateSource(); });
517
518 machineManufacturerSource.setMode(1);
519 machineModelSource.setMode(2);
520 machineModelFilter = machineModelSource.setFilter<MachineModelFilter>();
521
522 apertureSource.sort(apertureComparator);
523
524 connect(ui->manufacturerSearchInput, &DynamicComboBox::currentTextChanged,
525 [this](const QString &text) {
526 machineModelFilter->setManufacturer(text.toStdString());
527 });
528
529 ui->productTypeSearchInput->setDataSource(productSource);
530 ui->apertureSearchInput->setDataSource(apertureSource);
531 ui->thicknessSearchInput->setDataSource(topMaterialSource);
532 ui->backingThicknessSearchInput->setDataSource(bottomMaterialSource);
533 ui->manufacturerSearchInput->setDataSource(machineManufacturerSource);
534 ui->modelSearchInput->setDataSource(machineModelSource);
535 ui->deckSearchInput->setDataSource(machineDeckSource);
536}

763unsigned MainMenu::getValidRequestCode() const {
764 std::vector<unsigned> codes;
765
766 for (std::pair<unsigned, DrawingResponseMode> code : drawingResponseActions) {
767 codes.push_back(code.first);
768 }
769 for (unsigned i = 0; i < codes.size(); i++) {
770 unsigned target = codes[i];
771 while (target < codes.size() && target != codes[target]) {
772 unsigned temp = codes[target];
773 codes[target] = target;
774 target = temp;
775 }
776 }
777 for (unsigned i = 0; i < codes.size(); i++) {
778 if (codes[i] != i) {
779 return i;
780 }
781 }
782 return codes.size();
783}

1032void MainMenu::processDrawings() {
1033 drawingReceivedQueueMutex.lock();
1034 while (!drawingReceivedQueue.empty()) {
1035 DrawingRequest *request = drawingReceivedQueue.front();
1036 drawingReceivedQueue.pop();
1037
1038 switch (drawingResponseActions[request->responseEchoCode]) {
1039 case OPEN_DRAWING_VIEW_TAB:
1040 if (request->drawingData->loadWarning(
1041 Drawing::LoadWarning::LOAD_FAILED)) {
1042 QMessageBox::about(this, "Drawing Load Failed",
1043 "Attempt to load this drawing from "
1044 "the database failed.");
1045 } else {
1046 DrawingViewWidget *drawingView =
1047 new DrawingViewWidget(request->drawingData.value(), ui->mainTabs);
1048 Drawing &drawing = request->drawingData.value();
1049
1050 drawingView->setChangeDrawingCallback(
1051 [this,
1052 drawing](AddDrawingPageWidget::AddDrawingMode mode) mutable {
1053 bool automatic;
1054 if (drawing.drawingNumber().front() != 'M') {
1055 automatic = true;
1056 } else {
1057 automatic = false;
1058 }
1059 if (mode == AddDrawingPageWidget::CLONE_DRAWING) {
1060 if (automatic)
1061 drawing.setDrawingNumber(nextAutomaticDrawingNumber);
1062 else
1063 drawing.setDrawingNumber(nextManualDrawingNumber);
1064 drawing.setDate(Date::today());
1065 }
1066 AddDrawingPageWidget *addDrawingPage = new AddDrawingPageWidget(
1067 drawing, mode, automatic, ui->mainTabs);
1068 addDrawingPage->setUserEmail(clientEmailAddress);
1069 addDrawingPage->setConfirmationCallback(
1070 [this](const Drawing &drawing, bool force) {
1071 std::unique_ptr<DrawingInsert> insert(
1072 new DrawingInsert());
1073 insert->drawingData = drawing;
1074
1075 insert->setForce(force);
1076
1077 insert->responseEchoCode = getValidInsertCode();
1078 drawingInserts[insert->responseEchoCode] = &drawing;
1079
1080 unsigned bufferSize = insert->serialisedSize() + 1;
1081 void *buffer = malloc(bufferSize);
1082 insert->serialise(buffer);
1083
1084 client->addMessageToSendQueue(buffer, bufferSize);
1085 free(buffer);
1086 });
1087 switch (mode) {
1088 case AddDrawingPageWidget::CLONE_DRAWING:
1089 ui->mainTabs->addTab(addDrawingPage, tr("Clone Drawing"));
1090 break;
1091 case AddDrawingPageWidget::EDIT_DRAWING:
1092 ui->mainTabs->addTab(addDrawingPage, tr("Edit Drawing"));
1093 break;
1094 default:
1095 break;
1096 }
1097 ui->mainTabs->setCurrentWidget(addDrawingPage);
1098 });
1099
1100 ui->mainTabs->addTab(
1101 drawingView,
1102 tr((request->drawingData->drawingNumber() + " Details").c_str()));
1103 ui->mainTabs->setCurrentWidget(drawingView);
1104 }
1105 break;
1106 }
1107
1108 drawingResponseActions.erase(request->responseEchoCode);
1109 }
1110 drawingReceivedQueueMutex.unlock();
1111}
