7ExpandingWidget::ExpandingWidget(const QString &title, QWidget *parent)
8 : QWidget(parent) {

19 headerFrame->setFrameShadow(QFrame::Sunken);
20 headerFrame->setSizePolicy(QSizePolicy::Expanding, QSizePolicy::Maximum);

24 contentArea->setSizePolicy(QSizePolicy::Expanding, QSizePolicy::Fixed);
25 contentArea->setMaximumHeight(0);

27
28 mainLayout = new QGridLayout(this);
29 mainLayout->setVerticalSpacing(0);
30 mainLayout->setContentsMargins(0, 0, 0, 0);
31
32 unsigned row = 0;
33 mainLayout->addWidget(expandButton, row, 0, 1, 1, Qt::AlignLeft);
34 mainLayout->addWidget(headerFrame, row++, 2, 1, 1);
35 mainLayout->addWidget(contentArea, row, 0, 1, 3);
36 setLayout(mainLayout);
37
38 connect(expandButton, &QToolButton::clicked, [this](bool checked) {
39 expandButton->setArrowType(checked ? Qt::ArrowType::DownArrow : Qt::ArrowType::UpArrow);
40
41 const unsigned collapsedHeight = expandButton->sizeHint().height();
42 unsigned contentHeight = contentArea->layout()->sizeHint().height();
43
44 if (checked) {
45 setMinimumHeight(contentHeight + collapsedHeight);
46 setMaximumHeight(contentHeight + collapsedHeight);
47 contentArea->setMaximumHeight(contentHeight);
48 } else {
49 setMinimumHeight(collapsedHeight);
50 setMaximumHeight(collapsedHeight);
51 contentArea->setMaximumHeight(0);
52 }
53 });
