329std::vector<std::filesystem::path> DrawingViewWidget::punchProgramPathForDrawing(const std::string &drawingNumber) {
330 std::filesystem::path path = PUNCH_PDF_LOCATION;
331 std::string folder;
332
333 size_t index = 0;
334 while (index < drawingNumber.size()) {
335 if (std::isalpha(drawingNumber.at(index))) {
336 folder += drawingNumber.at(index++);
337 } else {
338 break;
339 }
340 }
341 if (folder == "M") {
342 folder = "2. Manuals";
343 } else if (folder.size() == 1) {
344 folder = "1" + folder;
345 }
346
347 std::vector<std::filesystem::path> out;
348 path /= folder;
349 for (const auto& entry : std::filesystem::directory_iterator(path)) {
350 if (std::filesystem::is_regular_file(entry.path()) && std::regex_match(entry.path().filename().string(), std::regex(drawingNumber + ".*\\.pdf"))) {
351 out.push_back(entry);
352 }
353 }
354
355 return out;
356};

359void DrawingViewWidget::updateTotals() const {
360 float totalCost = 0;
361 for (QLineEdit *edit : subtotalEdits) {
362 totalCost += edit->text().toFloat();
363 }
364 totalCost += extraBox->text().toFloat();
365 subtotal_lineEdit->setText(QString::number(totalCost, 'f', 2));
366
367 unsigned total_time = 0;
368 for (QLineEdit* edit : labourEdits) {
369 total_time += edit->text().toUInt();
370 }
371 totalTimeEdit->setText(QString::number(total_time));
372 std::optional<float> maybe_labour_total_cost = get_time_cost(total_time);
373 float labour_total_cost =
374 maybe_labour_total_cost.has_value() ? *maybe_labour_total_cost : 0;
375 labour_cost_lineEdit->setText(QString::number(labour_total_cost, 'f', 2));
376
377 total_lineEdit->setText(QString::number(totalCost + labour_total_cost, 'f', 2));
378
379 sales_total->setText(QString::number((totalCost + labour_total_cost) * (1 + (sales_increase->text().toFloat()/100)), 'f', 2));

382QPdfDocument *DrawingViewWidget::getDoc(const QString& path, const QSize &size) {
383 QPdfDocument* oldDoc = new QPdfDocument();
384 oldDoc->load(path);
385 QByteArray *pdfData = new QByteArray();
386 QBuffer *buffer = new QBuffer(pdfData);
387 buffer->open(QIODevice::ReadWrite);
388 QPdfWriter writer(buffer);
389 QPageLayout layout;
390 layout.setPageSize(QPageSize::A4);
391 layout.setOrientation(QPageLayout::Orientation::Landscape);
392 writer.setPageLayout(layout);
393 writer.setResolution(300);
394 QPainter painter;
395 painter.begin(&writer);
396 QPdfDocumentRenderOptions options;
397 options.setRenderFlags(QPdfDocumentRenderOptions::RenderFlag::Annotations);
398 for (int i = 0; i < oldDoc->pageCount(); i++) {
399 QImage img = oldDoc->render(i, QSize(writer.width(), writer.height()),
400 options);
401 painter.drawImage(0, 0, img);
402 writer.newPage();
403 }
404 painter.end();
405 QPdfDocument *pdfDocument = new QPdfDocument();
406 pdfDocument->load(buffer);
407 buffer->close();
408 return pdfDocument;
409 }
