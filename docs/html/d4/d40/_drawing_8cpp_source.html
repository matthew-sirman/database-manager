7void writeAtBitOffset(void *value, size_t valueByteLength, void *target,
8 size_t bitOffset) {
9 size_t startIndex = bitOffset / 8;
10 byte offset = bitOffset % 8;
11 byte writeByte;
12
13 byte *valueBytes = (byte *)value, *targetBytes = (byte *)target + startIndex;
14
15 if (offset == 0) {
16 memcpy(targetBytes, value, valueByteLength);
17 return;
18 }
19
20 writeByte = *targetBytes;
21
22 for (unsigned i = 0; i < valueByteLength; i++) {
23 writeByte |= (byte)(valueBytes[i] << offset);
24 *targetBytes++ = writeByte;
25 writeByte = valueBytes[i] >> (8u - offset);
26 }
27
28 *targetBytes = writeByte;
29}

31void readFromBitOffset(void *data, size_t bitOffset, void *target,
32 size_t bitReadSize) {
33 size_t startIndex = bitOffset / 8, fullReadLength = bitReadSize / 8;
34 byte offset = bitOffset % 8, finalByteSize = bitReadSize % 8;
35 byte readByte;
36
37 byte *dataBytes = (byte *)data, *targetBytes = (byte *)target;
38
39 if (offset == 0) {
40 memcpy(target, dataBytes + startIndex, fullReadLength);
41
42 if (finalByteSize != 0) {
43 targetBytes[fullReadLength] = dataBytes[startIndex + fullReadLength] &
44 (0xFFu >> (8u - finalByteSize));
45 }
46
47 return;
48 }
49
50 for (unsigned i = 0; i < fullReadLength; i++) {
51 readByte = dataBytes[startIndex + i] >> offset;
52 readByte |= (byte)(dataBytes[startIndex + i + 1] << (8u - offset));
53 *targetBytes++ = readByte;
54 }
55
56 if (finalByteSize != 0) {
57 readByte = dataBytes[startIndex + fullReadLength] >> offset;
58 readByte |=
59 (byte)(dataBytes[startIndex + fullReadLength + 1] << (8u - offset));
60 *targetBytes = readByte & (0xFFu >> (8u - finalByteSize));
61 }

839Drawing::BuildWarning Drawing::checkDrawingValidity(unsigned exclusions) const {
840 if (!std::regex_match(__drawingNumber,
841 std::regex(drawingNumberRegexPattern)) &&
842 !(exclusions & INVALID_DRAWING_NUMBER)) {
843 return INVALID_DRAWING_NUMBER;
844 }
845 if (!DrawingComponentManager<Product>::validComponentHandle(productHandle) &&
846 !(exclusions & INVALID_PRODUCT)) {
847 return INVALID_PRODUCT;
848 }
849 if (__width <= 0 && !(exclusions & INVALID_WIDTH)) {
850 return INVALID_WIDTH;
851 }
852 if (__length <= 0 && !(exclusions & INVALID_LENGTH)) {
853 return INVALID_LENGTH;
854 }
855 if (!DrawingComponentManager<Material>::validComponentHandle(
856 topLayerThicknessHandle) &&
857 !(exclusions & INVALID_TOP_MATERIAL)) {
858 return INVALID_TOP_MATERIAL;
859 }
860 if (bottomLayerThicknessHandle.has_value()) {
861 if (!DrawingComponentManager<Material>::validComponentHandle(
862 bottomLayerThicknessHandle.value()) &&
863 !(exclusions & INVALID_BOTTOM_MATERIAL)) {
864 return INVALID_BOTTOM_MATERIAL;
865 }
866 }
867 if (strapHandle[0].has_value()) {
868 if (!DrawingComponentManager<Strap>::validComponentHandle(
869 strapHandle[0].value()) &&
870 !(exclusions & INVALID_STRAPS)) {
871 return INVALID_STRAPS;
872 }
873 }
874 if (!DrawingComponentManager<Aperture>::validComponentHandle(
875 apertureHandle) &&
876 !(exclusions & INVALID_APERTURE)) {
877 return INVALID_APERTURE;
878 }
879 if (std::accumulate(barSpacings.begin(), barSpacings.end(), 0.0f) !=
880 (__width - 60 * (((int)(sideIron(LEFT).type == SideIronType::B)) + ((int)(sideIron(RIGHT).type == SideIronType::B))))
881 &&
882 !(exclusions & INVALID_BAR_SPACINGS)) {
883 return INVALID_BAR_SPACINGS;
884 }
885 if (std::find(barWidths.begin(), barWidths.end(), 0.0f) != barWidths.end() &&
886 !(exclusions & INVALID_BAR_WIDTHS)) {
887 return INVALID_BAR_WIDTHS;

890 sideIronHandles[0]) &&

892 return INVALID_SIDE_IRONS;

894 if (!DrawingComponentManager<SideIron>::validComponentHandle(
895 sideIronHandles[1]) &&
896 !(exclusions & INVALID_SIDE_IRONS)) {
