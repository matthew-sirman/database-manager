7void DatabaseResponseHandler::onMessageReceived(void *&&message,
8 unsigned int messageSize) {
9 switch (getDeserialiseType(message)) {
10 case RequestType::REPEAT_TOKEN_REQUEST: {
11 uint256 token;
12
13 memcpy(&token, (unsigned char *)message + sizeof(RequestType),
14 sizeof(uint256));
15
16 repeatTokenCallback(token);
17
18 free(message);
19 break;
20 }
21 case RequestType::USER_EMAIL_REQUEST: {
22 unsigned char *buff = (unsigned char *)message + sizeof(RequestType);
23 unsigned char emailLength = *buff++;
24 std::string email = std::string((const char *)buff, emailLength);
25
26 emailReceivedCallback(email);
27 free(message);
28 break;
29 }
30 case RequestType::DRAWING_SEARCH_QUERY:
31 if (populateResultsModel) {
32 // resultsModel->sourceDataFromBuffer(std::move(message));
33 populateResultsModel(std::move(message));
34 }
35 break;
36 case RequestType::DRAWING_INSERT: {
37 DrawingInsert response = DrawingInsert::deserialise(std::move(message));
38 if (drawingInsertCallback) {
39 drawingInsertCallback(response.insertResponseCode,
40 response.responseEchoCode);
41 }
42 break;
43 }
44 case RequestType::SOURCE_PRODUCT_TABLE:
45 DrawingComponentManager<Product>::sourceComponentTable(std::move(message),
46 messageSize);
47 break;
48 case RequestType::SOURCE_BACKING_STRIPS_TABLE:
49 DrawingComponentManager<BackingStrip>::sourceComponentTable(
50 std::move(message), messageSize);
51 break;
52 case RequestType::SOURCE_APERTURE_TABLE:
53 DrawingComponentManager<Aperture>::sourceComponentTable(
54 std::move(message), messageSize);
55 break;
56 case RequestType::SOURCE_APERTURE_SHAPE_TABLE:
57 DrawingComponentManager<ApertureShape>::sourceComponentTable(
58 std::move(message), messageSize);
59 break;
60 case RequestType::SOURCE_MATERIAL_TABLE:
61 DrawingComponentManager<Material>::sourceComponentTable(
62 std::move(message), messageSize);
63 break;
64 case RequestType::SOURCE_EXTRA_PRICES_TABLE:
65 DrawingComponentManager<ExtraPrice>::sourceComponentTable(
66 std::move(message), messageSize);
67 break;
68 case RequestType::SOURCE_LABOUR_TIMES_TABLE:
69 DrawingComponentManager<LabourTime>::sourceComponentTable(
70 std::move(message), messageSize);
71 break;
72 case RequestType::SOURCE_POWDER_COATING_TABLE:
73 DrawingComponentManager<PowderCoatingPrice>::sourceComponentTable(
74 std::move(message), messageSize);
75 break;
76 case RequestType::SOURCE_SIDE_IRON_TABLE:
77 DrawingComponentManager<SideIron>::sourceComponentTable(
78 std::move(message), messageSize);
79 break;
80 case RequestType::SOURCE_SIDE_IRON_PRICES_TABLE:
81 DrawingComponentManager<SideIronPrice>::sourceComponentTable(
82 std::move(message), messageSize);
83 break;
84 case RequestType::SOURCE_MACHINE_TABLE:
85 DrawingComponentManager<Machine>::sourceComponentTable(std::move(message),
86 messageSize);
87 break;
88 case RequestType::SOURCE_MACHINE_DECK_TABLE:
89 DrawingComponentManager<MachineDeck>::sourceComponentTable(
90 std::move(message), messageSize);
91 break;
92 case RequestType::SOURCE_STRAPS_TABLE:
93 DrawingComponentManager<Strap>::sourceComponentTable(std::move(message),
94 messageSize);
95 break;
96 case RequestType::DRAWING_DETAILS:
97 if (drawingReceivedCallback) {
98 drawingReceivedCallback(
99 DrawingRequest::deserialise(std::move(message)));
100 }
101 break;
102 case RequestType::ADD_NEW_COMPONENT:
103 if (addComponentCallback) {
104 addComponentCallback(
105 ComponentInsert::deserialise(std::move(message)).responseCode);
106 }
107 break;
108 case RequestType::GET_NEXT_DRAWING_NUMBER:
109 if (nextDrawingResponseCallback) {
110 nextDrawingResponseCallback(
111 NextDrawing::deserialise(std::move(message)));
112 }
113 break;
114 case RequestType::CREATE_DATABASE_BACKUP:
115 if (backupResponseCallback) {
116 backupResponseCallback(
117 DatabaseBackup::deserialise(std::move(message)).responseCode);
118 }
119 break;
120 }
121}
