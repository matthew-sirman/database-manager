295template <typename T, typename... Us>
296AddComponentWindow<T, Us...>::AddComponentWindow(
297 Client* client, std::string comp,
298 std::array<std::string, sizeof...(Us)> names, QWidget* parent)
299 : client(client) {
300 window = new AddCompWindow(client, comp, parent);
301 layout = new QFormLayout(window->getLayoutContainer());
302 std::vector<QWidget*> widgets;
303 std::size_t index = 0;
304 (
305 [&]() {
306 QWidget* widget = nullptr;
307 if constexpr (DrawingComponentConcept<Us>) {
308 ComboboxComponentDataSource<Us> source;
309 std::get<ComboboxComponentDataSource<Us>>(components) =
310 std::move(source);
311 DrawingComponentManager<Us>::addCallback([this]() {
312 std::get<ComboboxComponentDataSource<Us>>(components)
313 .updateSource();
314 });
315 std::get<ComboboxComponentDataSource<Us>>(components).updateSource();
316 widget = Helper<Us>::addDrawingComponent(
317 this->layout, names[index++],
318 std::get<ComboboxComponentDataSource<Us>>(components));
319 } else if constexpr (IsOptionalDrawingComponent<Us>) {
320 ComboboxComponentDataSource<typename Us::value_type> source;
321 std::get<ComboboxComponentDataSource<typename Us::value_type>>(
322 components) = std::move(source);
323 DrawingComponentManager<typename Us::value_type>::addCallback(
324 [this]() {
325 std::get<ComboboxComponentDataSource<typename Us::value_type>>(
326 components)
327 .updateSource();
328 });
329 std::get<ComboboxComponentDataSource<typename Us::value_type>>(
330 components)
331 .updateSource();
332 widget = Helper<Us>::addOptionalDrawingComponent(
333 this->layout, names[index++],
334 std::get<ComboboxComponentDataSource<typename Us::value_type>>(
335 components));
336 } else if constexpr (std::is_enum_v<Us>) {
337 widget = Helper<Us>::addEnumToLayout(this->layout, names[index++]);
338 } else {
339 widget =
340 Helper<Us>::addComponentToLayout(this->layout, names[index++]);
341 }
342 widgets.push_back(widget);
343 }(),
344 ...);
345
346 std::function<void(int)>&& accept = [widgets, client](int result) {
347 switch ((QDialog::DialogCode)result) {
348 case QDialog::DialogCode::Accepted: {
349 std::size_t index = 0;
350 ComponentInsert insert;
351 insert.setComponentData<typename TypeMap<T>::insertType>(
352 {([widgets, &index]() {
353 if constexpr (std::is_enum_v<Us>)
354 return std::forward<typename Map<Us>::type>(
355 Helper<typename Map<Us>::type>::getEnum(widgets[index++]));
356 else if constexpr (IsDrawingComponent<Us>)
357 return std::forward<typename Map<Us>::type>(
358 Helper<typename Map<Us>::type>::getComponent(
359 widgets[index++], true));
360 else
361 return std::forward<typename Map<Us>::type>(
362 Helper<typename Map<Us>::type>::getComponent(
363 widgets[index++], false));
364 }())...});
365 unsigned bufferSize = insert.serialisedSize();
366 void* buffer = alloca(bufferSize);
367 insert.serialise(buffer);
368 client->addMessageToSendQueue(buffer, bufferSize);
369 break;
370 }
371 case QDialog::DialogCode::Rejected:
372 break;
373 };
374 };
375 window->getLayoutContainer()->setLayout(layout);
376 window->setAcceptCallback(std::move(accept));
377};
