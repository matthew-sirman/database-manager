3QPainterPath GroupGraphicsItem<Drawing::CentreHole, Aperture>::calculateBounds(const Drawing::CentreHole &hole) const {
4 Aperture ap = hole.aperture();
5 bool longitudinal, rounded;
6 if (ap.getShape().shape == "ST" || ap.getShape().shape == "SQ") {
7 rounded = false;
8 longitudinal = false;
9 }
10 else if (ap.getShape().shape == "SL") {
11 rounded = false;
12 longitudinal = true;
13 }
14 else if (ap.getShape().shape == "DIA" || ap.getShape().shape == "RT") {
15 rounded = true;
16 longitudinal = false;
17 }
18 else if (ap.getShape().shape == "RL") {
19 rounded = true;
20 longitudinal = true;
21 }
22 QPointF holePoint;
23 QPointF start(matBounds.left(), matBounds.top());
24 QPointF h(hole.pos.x, hole.pos.y);
25 QPointF apPoint = longitudinal ? QPointF(ap.width, ap.length) : QPointF(ap.length, ap.width);
26 float ratio = longitudinal ? matBounds.width() / matWidth : matBounds.height() / matLength;
27 QPointF boxTL = start + ratio * (h - 0.5 * apPoint);
28 QSizeF size((ap.width / matWidth) * matBounds.width(),
29 (ap.length / matLength) * matBounds.height());
30 if (!longitudinal) {
31 size = QSizeF(size.height(), size.width());
32 }
33 QRectF holeBounds(boxTL, size);
34 QPainterPath path;
35 path.addRect(holeBounds);
36 return path;
37};
