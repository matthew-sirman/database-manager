54

62// Serialise the ValueRange object into the buffer
63template <typename T>
64void ValueRange<T>::serialise(void *buffer) const {
65 // Get the buffer as a T* for direct writing
66 T *buff = (T *)buffer;
67 // Write the first sizeof(T) bytes to be the value of the lowerBound
68 // variable, and analogously for the second sizeof(T).
69 *buff++ = lowerBound;
70 *buff = upperBound;
71}
72
73// Deserialise and construct a ValueRange from the buffer
74template <typename T>
75ValueRange<T> ValueRange<T>::deserialise(void *buffer) {
76 // Get the buffer as a T* for direct reading
77 T *buff = (T *)buffer;
78 // Create the range object to return
79 ValueRange<T> range;
80 // Read from the buffer into the lower and upper bounds
81 range.lowerBound = *buff++;
82 range.upperBound = *buff;
83
84 // Return the constructed value range
85 return range;
86}
87
88// Method to get the number of bytes this object will take up in the buffer
89template <typename T>
90unsigned ValueRange<T>::serialisedSize() const {
91 // Simply return twice the sizeof(T)
92 return 2 * sizeof(T);
93}
