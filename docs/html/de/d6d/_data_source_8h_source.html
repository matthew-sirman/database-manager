69// Filter function implementation
70template<typename IType>
71inline bool SourceFilter<IType>::filter(IType element) const {
72 // First we check if the current element is the endpoint. If it is, we simply
73 // return true (else the iterator will attempt to keep finding the next available
74 // element)
75 if (element == __iterEndpoint) {
76 return true;
77 }
78
79 // If the element is a proper element in the set, we return the actual filter function of
80 // the element.
81 return __filter(element);
82}

258 // Sets the filter

297 // Gets the adapted data object for this iterator and stores it for return
298 DataHolder ret(**this);
299 // Increments the iterator once
300 iter++;
301 // If there is a filter
302 if (__filter) {
303 // Increment the filter until the element is valid

306 }

450// Implementation of begin function
451template<typename T, typename IType>
452DataSourceIterator<T, IType> DataSource<T, IType>::begin() const {
453 // If there is a filter
454 if (__filter) {
455 // Update the fitler such that it knows the correct endpoint
456 __filter->setEndpoint(__end);
457 // Return a data source iterator for the start with the appropriate adapter and filter.
458 return DataSourceIterator<T, IType>(__begin, __adapter, *__filter);
459 } else {
460 // Return a data source iterator for the start with the appropriate adapter, but no filter.
461 return DataSourceIterator<T, IType>(__begin, __adapter);
462 }

489// Implementation for the set filter function
490template<typename T, typename IType>
491template<typename Filter>
492Filter *DataSource<T, IType>::setFilter() {
493 // First we assert that this is a valid filter for the data source type. This can be done statically as the data source system
494 // is based on template types.
495 static_assert(std::is_base_of<SourceFilter<IType>, Filter>::value, "Filter must derive from SourceFilter<IType>");
496 // Delete the filter in case there was already one installed
497 delete __filter;
498 // We then construct the filter object
499 __filter = new Filter();
500 // We then attach an update callback which simply updates the source
501 __filter->attachFilterUpdateCallback([this]() { updateSource(); });
502 // We then manually assert that the source has updated so the filter can apply
503 updateSource();
504 // Finally we return the filter object cast to the derived type (which we know it is) so the user has
505 // access to its pointer.
506 return (Filter *)__filter;
507}

509// Implementation of remove filter function
510template<typename T, typename IType>
511inline void DataSource<T, IType>::removeFilter() {
512 // If the filter exists, delete it
513 if (__filter) {
514 delete __filter;
515 }
516 // Set the filter to nullptr so we know it has been removed
517 __filter = nullptr;
518 // Update the source to indicate that the filter is now gone
519 updateSource();
520}
