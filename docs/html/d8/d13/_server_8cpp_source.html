34void Server::initialiseServer(unsigned short serverPort) {
35 if (serverSocket.bound()) {
36 return;
37 }
38
39 guardTCPSocketCode(serverSocket.create(), *errorStream);
40 guardTCPSocketCode(serverSocket.setNonBlocking(), *errorStream);
41 guardTCPSocketCode(serverSocket.bindToAddress(serverPort), *errorStream);
42
43 guardTCPSocketCode(serverSocket.beginListen(), *errorStream);
44
45 serverSocket.setAcceptCallback([this](TCPSocket &&s) {
46 std::future<void> acceptAsync = std::async(
47 std::launch::async, &Server::acceptClient, this, std::ref(s));
48
49 if (acceptAsync.wait_for(std::chrono::milliseconds(60000)) !=
50 std::future_status::ready) {
51 s.closeSocket();
52 }
53 });
54
55 Logger::log("Server initialised");
56}

58void Server::startServer() {
59 Logger::log("Server started successfully");
60
61 // Asynchronous call to console read to make console inputs non-blocking
62 std::future<std::string> nonBlockingInput =
63 std::async(std::launch::async, getNonBlockingInput);
64
65 // Variables to hold the start and end times of each update frame (for
66 // calculating frame delta time)
67 std::chrono::time_point<std::chrono::system_clock> start, end;
68
69 unsigned cycle = 0;
70
71 while (true) {
72 // Get the start time of this frame
73 start = std::chrono::system_clock::now();
74
75 // Check if any client is trying to connect to the server
76 switch (serverSocket.tryAccept(true)) {
77 case ERR_ACCEPT:
78 Logger::logError("Failed to accept client", -1, "", false);
79 case SOCKET_SUCCESS:
80 Logger::log("Accepted a new client");
81 case S_NO_DATA:
82 default:
83 break;
84 }
85
86 // For every client waiting for authentication, try to authenticate
87 // them.
88 for (int ic = 0; ic < waitingClients.size(); ic++) {
89 if (tryAuthenticateClient(*waitingClients[ic])) {
90 waitingClients.erase(waitingClients.begin() + ic);
91 }
92 }
93
94 // Check all connected client channels for incoming messages
95 for (int ic = 0; ic < connectedClients.size(); ic++) {
96 ClientData &connectedClient = *connectedClients[ic];
97
98 // First, check if the client socket connection is dead
99 /*if (connectedClient.clientSocket.dead()) {
100 *logStream << "Client " << connectedClient.clientEmail << "
101 disconnected." << std::endl;
102 connectedClient.clientSocket.closeSocket();
103 connectedClients.erase(connectedClients.begin() + ic);
104 handleMap.erase(connectedClient.handle.clientID);
105 }
106
107 EncryptedNetworkMessage message;
108 // If we haven't received a message, continue
109 if (connectedClient.clientSocket.receiveMessage(message,
110 MessageProtocol::AES_MESSAGE) != SOCKET_SUCCESS) { continue;
111 }*/
112
113 EncryptedNetworkMessage message;
114
115 switch (connectedClient.clientSocket.receiveMessage(
116 message, MessageProtocol::AES_MESSAGE)) {
117 case SOCKET_SUCCESS:
118 break;
119 case ERR_SOCKET_DEAD:
120 lockLog {
121 *ss << "Client " << connectedClient.clientEmail
122 << " disconnected (timed out).";
123 Logger::log();
124 }
125
126 connectedClient.clientSocket.closeSocket();
127 delete &connectedClient;
128 connectedClients.erase(connectedClients.begin() + ic);
129 handleMap.erase(connectedClient.handle.clientID);
130 continue;
131 case SOCKET_DISCONNECTED:
132 switch (*((DisconnectCode *)message.getMessageData())) {
133 case DisconnectCode::CLIENT_EXIT:
134 lockLog {
135 *ss
136 << "Client " << connectedClient.clientEmail
137 << " disconnected.";
138 Logger::log();
139 }
140 break;
141 default:
142 lockLog {
143 *ss
144 << "Client " << connectedClient.clientEmail
145 << " disconnected (with error code).";
146 Logger::log();
147 }
148 break;
149 }
150 connectedClient.clientSocket.closeSocket();
151 delete &connectedClient;
152 connectedClients.erase(connectedClients.begin() + ic);
153 handleMap.erase(connectedClient.handle.clientID);
154 continue;
155 default:
156 continue;
157 }
158
159 // If the message received successfully
160 if (!message.error()) {
161 // Get the message and decrypt it with this client's key
162 uint8 *decryptedMessage = (uint8 *)message.decryptMessageData(
163 connectedClient.clientSessionKey);
164 uint64 messageToken = *((uint64 *)decryptedMessage);
165 // We create a rvalue reference to indicate ownership, and we
166 // fill it with the message - the token.
167 void *&&msg = malloc(message.getMessageSize() - sizeof(uint64));
168 std::memmove(msg, decryptedMessage + sizeof(uint64),
169 message.getMessageSize() - sizeof(uint64));
170 free(decryptedMessage);
171 // If the message starts with the client's secret session token,
172 // the message is valid
173 if (messageToken == connectedClient.clientSessionToken) {
174 // Assuming we have set an appropriate handler, call it with
175 // the decrypted message
176 if (requestHandler) {
177 requestHandler->onMessageReceived(
178 *this, connectedClient.handle, std::move(msg),
179 message.getMessageSize() - sizeof(uint64));
180 }
181 }
182 }
183 }
184
185 // Send any messages in the send queue
186 sendQueueMutex.lock();
187 while (!sendQueue.empty()) {
188 std::pair<unsigned, NetworkMessage *> message = sendQueue.front();
189 handleMap[message.first]->clientSocket.sendMessage(*message.second);
190 sendQueue.pop();
191 delete message.second;
192 }
193 sendQueueMutex.unlock();
194
195 // Every n cycles
196 if (cycle % heartBeatCycles == 0) {
197 for (ClientData *connectedClient : connectedClients) {
198 connectedClient->clientSocket.heartbeat();
199 }
200 }
201
202 if (nonBlockingInput.wait_for(std::chrono::milliseconds(0)) ==
203 std::future_status::ready) {
204 std::string input = nonBlockingInput.get();
205
206 if (input == "quit" || input == "exit") {
207 lockLog {
208 *ss << "Server closed" << std::endl;
209 Logger::log();
210 }
211 break;
212 }
213 if (input == "list users") {
214 for (ClientData *connectedClient : connectedClients) {
215 std::cout << "Client: " << connectedClient->clientEmail << std::endl;
216 }
217 }
218
219 nonBlockingInput = std::async(std::launch::async, getNonBlockingInput);
220 }
221
222 // If the refresh rate is either negative or 0 it is invalid, so raise
223 // an error (otherwise there is the risk of division by 0 or running a
224 // while loop at maximum speed causing the process to overload the CPU!)
225 if (refreshRate <= 0) {
226 ERROR_TO("Refresh rate is <= 0. This is not allowed", *errorStream)
227 }
228
229 // Get the end time of this frame
230 end = std::chrono::system_clock::now();
231
232 // Calculate how long this frame took to execute
233 std::chrono::duration<double> frameElapsed = end - start;
234
235 // Calculate how much extra time must be waited to have the frame take
236 // the correct total time
237 double delta = (1.0 / refreshRate) - frameElapsed.count();
238
239 // If we have time to wait, wait it, otherwise don't (can't wait a
240 // negative time!)
241 if (delta > 0) {
242 // Sleep function in microseconds, so multiply by 1e6
243 // usleep(delta * 1e6);
244 std::this_thread::sleep_for(
245 std::chrono::microseconds((long long)(delta * 1e6)));
246 }
247
248 cycle++;
249 }
250}

377void Server::acceptClient(TCPSocket &clientSocket) {
378 // PROTOCOL:
379 // Client and server send each other their respective public keys
380 // Client sends a random challenge to the server, encrypted under the
381 // server's public key. The server decrypts this message and generates a
382 // random session token for this client along with a random AES session key
383 // for this client. It also generates its own random number used once to be
384 // used in the JWT request. It then concatenates the challenge, session key,
385 // token and new nonce, and signs them all under the digital signature key
386 // for the server. Then, it encrypts this signed value under the client's
387 // public key and sends it back to the client.
388 // At this point, neither the client nor the server have authenticated
389 // themselves to each other. The server notes that this client is currently
390 // not authenticated and waits for proof of who they are. The client
391 // receives the encrypted and signed data from the server. It first decrypts
392 // using its private key to see the message, and then decrypts the signature
393 // with the server's public signature key. It expects to see the number it
394 // generated. If it does not see the number, then the server has failed to
395 // authenticate itself and the client terminates the connection; it is not
396 // communicating with the correct server. If it does see its challenge, the
397 // server must have signed it, so the client knows they are talking to the
398 // server. They now have an AES session key in common with the server, and
399 // they also have their session token allowing them to communicate. Next, in
400 // order to access any resource from the server, the client must also
401 // authenticate itself to the server. It knows by this point that the server
402 // is trusted. The client first gathers a JWT from the Microsoft graph
403 // authentication network. This JWT will be signed by Microsoft for a
404 // specific account. The data in the JWT cannot be tampered with, else the
405 // signature will be invalid. The client then constructs a message
406 // containing first its token, followed by the JWT it has received. It then
407 // encrypts this under the symmetric AES key and sends to the server.
408 // The server then decrypts this message, and if it sees the client's token,
409 // it knows that this client must have sent the message, as no one else
410 // knows the token number. The server then authenticates the JWT. If the
411 // authentication is successful, the client has successfully authenticated
412 // themselves to the server and they may now access resources. Otherwise,
413 // the server informs the client that they have failed, and terminates the
414 // connection.
415 //
416 // 1. C -> S: KC
417 // 2. S -> C: KS
418 // 3. C -> S: {NC}_KS
419 // 4. S -> C: {{NC, NS, K, T}_SIG}_KC
420 // if NC invalid:
421 // client terminates connection
422 // end
423 // 5. C -> S: {T, JWT}_K
424 // if JWT invalid:
425 // S -> C: ERROR
426 // server terminates connection
427 //
428 // Both communicate with AES key from here, with messages starting with
429 // token
430
431 // 1: Receive client's public key
432 RSAKeyPair::Public clientKey;
433 NetworkMessage clientKeyMessage;
434
435 // If the client socket doesn't send us a message, close the connection and
436 // return
437 if (clientSocket.receiveMessage(
438 clientKeyMessage, MessageProtocol::KEY_MESSAGE) != SOCKET_SUCCESS) {
439 ConnectionResponse failResponse = ConnectionResponse::FAILED;
440 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
441 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
442 clientSocket.sendMessage(failedMessage);
443
444 clientSocket.closeSocket();
445 return;
446 }
447
448 // If this message was in any way erroneous, close the connection and return
449 if (clientKeyMessage.error()) {
450 ConnectionResponse failResponse = ConnectionResponse::FAILED;
451 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
452 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
453 clientSocket.sendMessage(failedMessage);
454
455 clientSocket.closeSocket();
456 return;
457 }
458
459 memcpy(&clientKey, clientKeyMessage.getMessageData(),
460 clientKeyMessage.getMessageSize());
461
462 // 2: Send server's public key
463 if (clientSocket.sendMessage(
464 NetworkMessage(&serverKey.publicKey, sizeof(RSAKeyPair::Public),
465 MessageProtocol::KEY_MESSAGE)) != SOCKET_SUCCESS) {
466 ConnectionResponse failResponse = ConnectionResponse::FAILED;
467 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
468 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
469 clientSocket.sendMessage(failedMessage);
470
471 clientSocket.closeSocket();
472 return;
473 }
474
475 // 3: Receive the client's random challenge, encrypted under server's public
476 // key
477 uint2048 encryptedChallenge;
478 NetworkMessage challengeMessage;
479
480 if (clientSocket.waitForMessage(
481 challengeMessage, MessageProtocol::RSA_MESSAGE) != SOCKET_SUCCESS) {
482 ConnectionResponse failResponse = ConnectionResponse::FAILED;
483 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
484 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
485 clientSocket.sendMessage(failedMessage);
486
487 clientSocket.closeSocket();
488 return;
489 }
490
491 // If this message was in any way erroneous, close the connection and return
492 if (challengeMessage.error()) {
493 ConnectionResponse failResponse = ConnectionResponse::FAILED;
494 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
495 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
496 clientSocket.sendMessage(failedMessage);
497
498 clientSocket.closeSocket();
499 return;
500 }
501
502 memcpy(&encryptedChallenge, challengeMessage.getMessageData(),
503 challengeMessage.getMessageSize());
504
505 uint2048 decryptedChallenge =
506 decrypt(encryptedChallenge, serverKey.privateKey);
507
508 // Check that the challenge is valid. If it isn't a uint64 padded by 0's, is
509 // is not correct; terminate the connection
510 uint2048 mask = 0xFFFFFFFFFFFFFFFFull;
511
512 if ((decryptedChallenge & (~mask)) != 0) {
513 ConnectionResponse failResponse = ConnectionResponse::FAILED;
514 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
515 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
516 clientSocket.sendMessage(failedMessage);
517
518 clientSocket.closeSocket();
519 return;
520 }
521
522 // 4: Send signed challenge, nonce, session key and token
523 AESKey clientSessionKey = generateAESKey();
524 uint64 clientSessionToken;
525 uint32 clientNonce;
526 CryptoSafeRandom::random(&clientSessionToken, sizeof(uint64));
527 CryptoSafeRandom::random(&clientNonce, sizeof(uint32));
528
529 uint2048 response;
530 uint8 *responseBuffer = (uint8 *)&response;
531 memcpy(responseBuffer, &decryptedChallenge, sizeof(uint64));
532 memcpy(responseBuffer + sizeof(uint64), &clientNonce, sizeof(uint32));
533 memcpy(responseBuffer + sizeof(uint64) + sizeof(uint32), &clientSessionKey,
534 sizeof(AESKey));
535 memcpy(responseBuffer + sizeof(uint64) + sizeof(uint32) + sizeof(AESKey),
536 &clientSessionToken, sizeof(uint64));
537
538 uint2048 signedEncryptedResponse =
539 encrypt(sign(response, serverSignature.privateKey), clientKey);
540
541 if (clientSocket.sendMessage(
542 NetworkMessage(signedEncryptedResponse.rawData(), sizeof(uint2048),
543 MessageProtocol::RSA_MESSAGE)) != SOCKET_SUCCESS) {
544 ConnectionResponse failResponse = ConnectionResponse::FAILED;
545 NetworkMessage failedMessage(&failResponse, sizeof(ConnectionResponse),
546 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
547 clientSocket.sendMessage(failedMessage);
548
549 clientSocket.closeSocket();
550 return;
551 }
552
553 // After we have received all the protocol data, set this socket to be non
554 // blocking
555 clientSocket.setNonBlocking();
556
557 std::vector<unsigned> handles;
558
559 for (std::pair<unsigned, ClientData *> handle : handleMap) {
560 handles.push_back(handle.first);
561 }
562 for (unsigned i = 0; i < handles.size(); i++) {
563 unsigned target = handles[i];
564 while (target < handles.size() && target != handles[target]) {
565 unsigned temp = handles[target];
566 handles[target] = target;
567 target = temp;
568 }
569 }
570 unsigned handleID = handles.size();
571 for (unsigned i = 0; i < handles.size(); i++) {
572 if (handles[i] != i) {
573 handleID = i;
574 break;
575 }
576 }
577
578 ClientData *client = new ClientData(handleID, clientSocket, clientSessionKey,
579 clientSessionToken, clientNonce);
580
581 waitingClients.push_back(client);
582
583 handleMap[handleID] = client;
584}

586bool Server::tryAuthenticateClient(ClientData &clientData) {
587 // Attempt to receive a message from the client. If they aren't sending
588 // anything, or they send an erroneous message, just return false.
589 EncryptedNetworkMessage authMessage;
590 if (clientData.clientSocket.receiveMessage(
591 authMessage, MessageProtocol::AES_MESSAGE) == S_NO_DATA) {
592 return false;
593 }
594
595 if (authMessage.error()) {
596 return false;
597 }
598
599 unsigned char *authMessageBuff =
600 (unsigned char *)authMessage.decryptMessageData(
601 clientData.clientSessionKey);
602 AuthMode authMode = *((AuthMode *)authMessageBuff);
603 authMessageBuff += sizeof(AuthMode);
604
605 switch (authMode) {
606 case AuthMode::JWT: {
607 // Get a string of the response
608 std::string jwtResponse =
609 std::string((const char *)authMessageBuff,
610 authMessage.getMessageSize() - sizeof(AuthMode));
611
612 nlohmann::json claims;
613
614 // Attempt to authenticate this JWT
615 MicrosoftAccountAuthState status =
616 authenticateMicrosoftAccount(jwtResponse, CLIENT_APPLICATION_ID,
617 clientData.clientAuthNonce, "", claims);
618
619 // Check the status
620 switch (status) {
621 case AUTHENTICATED: {
622 // If the client successfully authenticated themselves,
623 // add them to the connected clients vector.
624 // They may now access server resources
625 clientData.clientEmail = claims["email"];
626 connectedClients.push_back(&clientData);
627 lockLog {
628 *ss << "Client " << clientData.clientEmail
629 << " successfully authenticated themselves.";
630 Logger::log();
631 }
632
633 clientData.access = ClientAccess::LIMITED;
634 ConnectionResponse successResponse = ConnectionResponse::SUCCESS;
635 if (claims.contains("roles")) {
636 if (claims["roles"].is_array()) {
637 auto it = std::find(claims["roles"].begin(),
638 claims["roles"].end(), "Access.Full");
639 if (it != claims["roles"].end()) {
640 clientData.access = ClientAccess::FULL;
641 successResponse = ConnectionResponse::SUCCESS_ADMIN;
642 }
643 }
644 }
645 NetworkMessage succeededMessage(
646 &successResponse, sizeof(ConnectionResponse),
647 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
648 clientData.clientSocket.sendMessage(succeededMessage);
649
650 return true;
651 }
652 case RECEIVED_ERRONEOUS_TOKEN:
653 lockLog {
654 *ss
655 << "Client failed to authenticate themselves: Bad JWT "
656 "(Erroneous Token). Terminating connection.";
657 Logger::log();
658 }
659 case NO_MATCHING_KEY:
660 lockLog {
661 *ss
662 << "Client failed to authenticate themselves: Bad JWT "
663 "(No matching key). Terminating connection.";
664 Logger::log();
665 }
666 case INVALID_TOKEN:
667 lockLog {
668 *ss
669 << "Client failed to authenticate themselves: Bad JWT "
670 "(Invalid token). Terminating connection.";
671 Logger::log();
672 }
673 case INVALID_SIGNATURE:
674 // If the authentication failed for any reason, the client
675 // did not authenticate themselves, so terminate their
676 // connection. If they wish to retry, they must reconnect to
677 // the server.
678 lockLog {
679 *ss
680 << "Client failed to authenticate themselves: Bad JWT "
681 "(Invalid signature). Terminating connection."
682 << std::endl;
683 Logger::log();
684 }
685 ConnectionResponse failResponse = ConnectionResponse::FAILED;
686 NetworkMessage failedMessage(
687 &failResponse, sizeof(ConnectionResponse),
688 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
689 clientData.clientSocket.sendMessage(failedMessage);
690
691 clientData.clientSocket.closeSocket();
692 break;
693 }
694 break;
695 }
696 case AuthMode::REPEAT_TOKEN:
697 uint256 repeatToken;
698
699 memcpy(&repeatToken, authMessageBuff, sizeof(uint256));
700
701 for (std::map<uint256,
702 std::pair<std::string, ClientAccess>>::const_iterator it =
703 repeatTokenMap.begin();
704 it != repeatTokenMap.end(); it++) {
705 if (it->first == repeatToken) {
706 std::pair<std::string, ClientAccess> info =
707 repeatTokenMap[repeatToken];
708 clientData.clientEmail = info.first;
709 clientData.access = info.second;
710 connectedClients.push_back(&clientData);
711 lockLog {
712 *ss
713 << "Client " << clientData.clientEmail
714 << " successfully authenticated themselves." << std::endl;
715 Logger::log();
716 }
717 ConnectionResponse successResponse = ConnectionResponse::SUCCESS;
718 if (clientData.access == ClientAccess::FULL) {
719 successResponse = ConnectionResponse::SUCCESS_ADMIN;
720 }
721 NetworkMessage succeededMessage(
722 &successResponse, sizeof(ConnectionResponse),
723 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
724 clientData.clientSocket.sendMessage(succeededMessage);
725
726 return true;
727 }
728 }
729
730 // There is no client with this repeat token
731 lockLog {
732 *ss
733 << "Client failed to authenticate themselves: Invalid "
734 "Repeat Token. Terminating connection.";
735 Logger::log();
736 }
737 ConnectionResponse failResponse = ConnectionResponse::FAILED;
738 NetworkMessage failedMessage(
739 &failResponse, sizeof(ConnectionResponse),
740 MessageProtocol::CONNECTION_RESPONSE_MESSAGE);
741 clientData.clientSocket.sendMessage(failedMessage);
742
743 clientData.clientSocket.closeSocket();
744 break;
745 }
746
747 return true;
748}

750// Old protocol. See above for fixed protocol
751// PROTOCOL: (this protocol might not work as the messages in steps 3 and 5 may
752// be too long to encrypt
753// asymmetrically. See the second protocol which avoids this issue)
754// Client and server send each other their respective public keys
755// Client sends a JWT to authenticate itself, along with a random
756// challenge, all encrypted under the server's public key. The server then
757// decrypts this, authenticates the JWT. If the JWT authentication fails, the
758// server sends an error to the client and refuses access. If the authentication
759// succeeds, the server signs the random challenge with its signing private key,
760// and sends the signed challenge and a session AES key, as well as a session
761// token for this client all encrypted under the client's public key back to the
762// client. The client decrypts the message it receives, and then verifies that
763// the server did in fact sign the challenge it send. Now, both the server and
764// client are authenticated to each other, and both have a common shared AES key
765// for communication. Additionally, there is now a token that both the server
766// and client know, so all messages should start with this token.
767// 1. C -> S: KC
768// 2. S -> C: KS
769// 3. C -> S: {JWT, NC}_KS
770// if not authenticated:
771// 4. S -> C: ERROR
772// terminate connection
773// else:
774// 5. S -> C: {{NC}_SIG, K, T}_KC
775// Both communicate with the session AES key K from here.
