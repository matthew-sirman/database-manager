67void DrawingPDFWriter::drawLabelAndField(QPainter &painter, double left,
68 double &top, const QString &label,
69 double labelWidth,
70 const QString &field,
71 double fieldWidth, double hOffset,
72 double vOffset) const {
73 QTextOption topLeftTextOption;
74 topLeftTextOption.setAlignment(Qt::AlignLeft | Qt::AlignTop);
75 topLeftTextOption.setWrapMode(QTextOption::WordWrap);
76
77 QRectF labelBounds =
78 QFontMetrics(painter.font())
79 .boundingRect(QRect(left, top, labelWidth, 0)
80 .adjusted(hOffset, vOffset, -hOffset, -vOffset),
81 Qt::TextWordWrap, label);
82 QRectF fieldBounds =
83 QFontMetrics(painter.font())
84 .boundingRect(QRect(left + labelWidth, top, fieldWidth, 0)
85 .adjusted(hOffset, vOffset, -hOffset, -vOffset),
86 Qt::TextWordWrap, field);
87
88 double height = MAX(labelBounds.height(), fieldBounds.height()) + 2 * vOffset;
89
90 painter.drawRect(QRectF(left, top, labelWidth, height));
91 painter.drawRect(QRectF(left + labelWidth, top, fieldWidth, height));
92 painter.drawText(labelBounds, label, topLeftTextOption);
93 painter.drawText(fieldBounds, field, topLeftTextOption);
94
95 top += height;
96}
