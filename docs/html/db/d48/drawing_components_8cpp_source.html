101std::string Aperture::apertureName() const {
102 ApertureShape shape =
103 DrawingComponentManager<ApertureShape>::getComponentByHandle(
104 apertureShapeID);
105
106 std::stringstream shapeName;
107
108 if (shape.shape == "Blank") {
109 shapeName << "Blank";
110 } else if (shape.shape == "SQ" || shape.shape == "DIA") {
111 shapeName << width << shape.shape;
112 } else if (shape.shape == "BOTH") {
113 shapeName << "ERROR!";
114 } else {
115 shapeName << width << shape.shape << length;
116 }
117
118 if (nibbleApertureId.has_value()) {
119 shapeName << " (Nibble using " +
120 DrawingComponentManager<Aperture>::findComponentByID(
121 *nibbleApertureId)
122 .apertureName() +
123 ")";
124 }
125
126 return shapeName.str();
127}

133bool Aperture::operator<(const Aperture &other) const {
134 if (this->apertureShapeID != other.apertureShapeID) {
135 const unsigned int ap1 =
136 ApertureShape::shapeOrder.at(this->apertureShapeID);
137 const unsigned int ap2 =
138 ApertureShape::shapeOrder.at(other.apertureShapeID);
139 return ap1 < ap2;
140 }
141 return this->width < other.width;
142}

170Material *Material::fromSource(unsigned char **buff) {
171 Material *material = new Material(*((unsigned *)*buff));
172 *buff += sizeof(unsigned);
173
174 material->hardness = *((unsigned short *)(*buff));
175 *buff += sizeof(unsigned short);
176
177 material->thickness = *((unsigned short *)(*buff));
178 *buff += sizeof(unsigned short);
179
180 unsigned char nameSize = *(*buff)++;
181
182 material->materialName = std::string((const char *)*buff, nameSize);
183 *buff += nameSize;
184
185 unsigned char priceElements = *(*buff)++;
186 for (unsigned char i = 0; i < priceElements; i++) {
187 unsigned material_price_id = *((unsigned *)(*buff));
188 *buff += sizeof(unsigned);
189 float width = *((float *)(*buff));
190 *buff += sizeof(float);
191 float length = *((float *)(*buff));
192 *buff += sizeof(float);
193 float price = *((float *)(*buff));
194 *buff += sizeof(float);
195 MaterialPricingType pricingType = *((MaterialPricingType *)(*buff));
196 *buff += sizeof(MaterialPricingType);
197 material->materialPrices.push_back(
198 {material_price_id, width, length, price, pricingType});
199 }
200 std::function<bool(const MaterialPrice &, const MaterialPrice &)>
201 materialPriceComparator =
202 [](const MaterialPrice &t1, const MaterialPrice &t2) {
203 if (std::get<1>(t1) != std::get<1>(t2)) {
204 return std::get<1>(t1) < std::get<1>(t2);
205 }
206 return std::get<3>(t1) < std::get<3>(t2);
207 };
208 sort(material->materialPrices.begin(), material->materialPrices.end(),
209 materialPriceComparator);
210
211 return material;
212}

266 extraPrice);
