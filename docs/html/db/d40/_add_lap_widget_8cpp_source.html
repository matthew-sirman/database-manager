7AddLapWidget::AddLapWidget(QGraphicsScene *scene, QRectF hoverBounds, DimensionLine::Orientation orientation,
8 const std::optional<Drawing::Lap> &currentLap, const QColor &unusedColour,
9 const QColor &unusedHighlightColour,
10 const QColor &usedColour) {
11 this->hoverBounds = hoverBounds;
12 this->unusedColour = unusedColour;
13 this->unusedHighlightColour = unusedHighlightColour;
14 this->usedColour = usedColour;
15 this->orientation = orientation;
16
17 hovering = false;
18
19 if (currentLap.has_value()) {
20 lap = currentLap.value();
21 } else {
22 lap = Drawing::Lap();
23 }
24 used = false;
25
26 setAcceptHoverEvents(true);
27
28 widthInput = new QDoubleSpinBox();
29 widthInput->setDecimals(1);
30 widthInput->setMinimum(0);
31 widthInput->setMaximum(32767);
32 widthInput->setValue(lap.width);
33
34 QWidget::connect(widthInput, &QDoubleSpinBox::editingFinished, [this]() {
35 lap.width = (float) widthInput->value();
36 if (lapChangedCallback) {
37 lapChangedCallback(lap);
38 }
39 });
40
41 widthProxy = scene->addWidget(widthInput);
42 widthProxy->setZValue(1);
43
44 QRectF dimensionPosition;
45 switch (orientation) {
46 case DimensionLine::VERTICAL:
47 dimensionPosition = QRectF(hoverBounds.left(),
48 hoverBounds.bottom() -
49 hoverBounds.height() * (verticalEditPosition + editSize / 2),
50 hoverBounds.width(), hoverBounds.height() * editSize);
51 widthDimension = new WidgetDimensionLine(dimensionPosition, DimensionLine::HORIZONTAL, widthProxy);
52 break;
53 case DimensionLine::HORIZONTAL:
54 dimensionPosition = QRectF(hoverBounds.left() + hoverBounds.width() * (horizontalEditPosition + editSize / 2),
55 hoverBounds.top(),
56 hoverBounds.width() * editSize, hoverBounds.height());
57 widthDimension = new WidgetDimensionLine(dimensionPosition, DimensionLine::VERTICAL, widthProxy);
58 break;
59 }
60 widthDimension->setZValue(0.5);
61
62 widthProxy->setVisible(used);
63 widthDimension->setVisible(used);
64
65 scene->addItem(widthDimension);
66
67 DrawingComponentManager<Material>::addCallback([this]() { materialSource.updateSource(); });
68 materialSource.updateSource();
69
70 setToolTip("Click to add overlap");
71}

176void AddLapWidget::contextMenuEvent(QGraphicsSceneContextMenuEvent *event) {
177 if (used) {
178 QMenu *menu = new QMenu();
179
180 QMenu *materialSubMenu;
181
182 switch (lap.attachmentType) {
183 case LapAttachment::INTEGRAL:
184 materialSubMenu = menu->addMenu("Make Bonded");
185 break;
186 case LapAttachment::BONDED:
187 menu->addAction("Make Integral", [this]() {
188 lap.attachmentType = LapAttachment::INTEGRAL;
189 if (lapChangedCallback) {
190 lapChangedCallback(lap);
191 }
192 });
193
194 materialSubMenu = menu->addMenu("Material");
195 break;
196 }
197
198 for (const ComboboxDataElement &mat : materialSource) {
199 QAction *action = materialSubMenu->addAction(mat.text.c_str(), [this, mat]() {
200 if (mat.index.has_value()) {
201 lap.setMaterial(mat.index.value());
202 lap.attachmentType = LapAttachment::BONDED;
203 if (lapChangedCallback) {
204 lapChangedCallback(lap);
205 }
206 }
207 });
208
209 if (mat.index.has_value()) {
210 if (mat.index.value() == lap.material().handle()) {
211 QFont font = action->font();
212 font.setBold(true);
213 font.setUnderline(true);
214 action->setFont(font);
215 }
216 }
217 }
218
219 menu->addAction("Remove overlap",
220 [this]() { setUsed(false); }, Qt::Key_Delete);
221 menu->popup(event->screenPos());
222 QWidget::connect(menu, &QMenu::triggered, [menu](QAction *) { menu->deleteLater(); });
223 }
224 QGraphicsItem::contextMenuEvent(event);
225}
