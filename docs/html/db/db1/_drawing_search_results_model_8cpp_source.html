22QVariant DrawingSearchResultsModel::data(const QModelIndex &index, int role) const {
23 if (role == Qt::DisplayRole) {
24 DrawingSummary summary = summaries[index.row()];
25
26 switch (index.column()) {
27 case 0:
28 return QString(summary.drawingNumber.c_str());
29 case 1:
30 return QString(summary.summaryString().c_str());
31 case 2: {
32 if (summary.barSpacingCount() == 1) {
33 return QString("No bars");
34 }
35 std::stringstream spacingsString;
36 std::vector<float> spacings = summary.barSpacings();
37 for (std::vector<float>::const_iterator it = spacings.begin(); it != spacings.end(); it++) {
38 spacingsString << *it;
39 if (it != spacings.end() - 1) {
40 spacingsString << "+";
41 }
42 }
43 return QString(spacingsString.str().c_str());
44 }
45 default:
46 break;
47 }
48 }

50 return QVariant();

72void DrawingSearchResultsModel::sourceDataFromBuffer(void*&& buffer) {
73 unsigned char *buff = (unsigned char *)buffer;
74 buff += sizeof(RequestType);
75
76 DrawingSummaryCompressionSchema schema = *((DrawingSummaryCompressionSchema *) buff);
77 buff += sizeof(DrawingSummaryCompressionSchema);
78
79 unsigned recordCount = *((unsigned *)buff);
80 buff += sizeof(unsigned);
81
82 if (summaries.size() != 0) {
83 beginRemoveRows(QModelIndex(), 0, (int)summaries.size() - 1);
84 removeRows(0, (int)summaries.size() - 1);
85 endRemoveRows();
86 }
87
88 summaries.clear();
89
90 for (unsigned i = 0; i < recordCount; i++) {
91 unsigned size;
92 DrawingSummary summary = schema.uncompressSummary(buff, size);
93
94 buff += size;
95
96 summaries.push_back(summary);
97 }
98
99 if (summaries.size() != 0) {
100 beginInsertRows(QModelIndex(), 0, (int)recordCount - 1);
101 insertRows(0, (int)recordCount - 1);
102 endInsertRows();
103 }
104
105 free(buffer);
106 emit dataChanged(index(0, 0), index(rowCount(), columnCount()));
107}
