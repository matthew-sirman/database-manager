166void AreaGraphicsItem<T, Ds...>::mousePressEvent(QGraphicsSceneMouseEvent* event) {
167 if (inspector) {
168 if (inspector->currentOwner() != inspectorAcquireID) {
169 inspectorAcquireID = inspector->acquire();
170 std::apply([this](ComboboxComponentDataSource<Ds>*...comboboxSources) {
171 AreaGraphicsItem<T, Ds...>::populateInspector(item, inspector, comboboxSources...);
172 }, componentSources);
173 }
174 }
175 QGraphicsItem::mousePressEvent(event);
176}

189template<typename T, DrawingComponentConcept ...Ds>
190void AreaGraphicsItem<T, Ds...>::contextMenuEvent(QGraphicsSceneContextMenuEvent* event) {
191 if (boundingBox.contains(event->pos())) {
192 QMenu* menu = new QMenu();
193
194 menu->addAction("Remove", [this]() {
195 if (inspectorAcquireID == inspector->currentOwner()) {
196 inspector->clear();
197 }
198 removeFunction();
199 }, Qt::Key_Delete);
200 menu->popup(event->screenPos());
201
202 QWidget::connect(menu, &QMenu::triggered, [menu](QAction*) { menu->deleteLater(); });
203 }
204
205 QGraphicsItem::contextMenuEvent(event);
