52void Inspector::addIntegerField(const QString &label, int &field, int min, int max) {
53 QSpinBox *fieldInput = new QSpinBox();
54 fieldInput->setMinimum(min);
55 fieldInput->setMaximum(max);
56 fieldInput->setValue(field);
57 QWidget::connect(fieldInput, &QSpinBox::editingFinished, [this, &field, fieldInput]() {
58 field = fieldInput->value();
59 std::for_each(updateTriggers.begin(), updateTriggers.end(), [](const std::function<void()> &trigger) { trigger(); });
60 });
61
62 contents->addRow(label, fieldInput);
63}
