30Client::ConnectionStatus Client::connectToServer(
31 const std::string &ipAddress, unsigned port,
32 const std::function<void(const std::string &)> &authStringCallback) {
33 if (!safeGuardTCPSocketCode(clientSocket.connectToServer(ipAddress, port))) {
34 clientSocket.closeSocket();
35 return ConnectionStatus::NO_CONNECTION;
36 }
37
38 // Protocol description found in the server accept method.
39 //
40 // 1. C -> S: KC
41 // 2. S -> C: KS
42 // 3. C -> S: {NC}_KS
43 // 4. S -> C: {{NC, NS, K, T}_SIG}_KC
44 // if NC invalid:
45 // client terminates connection
46 // end
47 // 5. C -> S: {T, JWT}_K
48 // if JWT invalid:
49 // S -> C: ERROR
50 // server terminates connection
51 // Both communicate with AES key from here, with messages starting with token
52
53 // 1: Send client's public key to server
54 clientSocket.sendMessage(NetworkMessage(&clientKey.publicKey,
55 sizeof(RSAKeyPair::Public),
56 MessageProtocol::KEY_MESSAGE));
57
58 // 2: Receive server's public key
59 RSAKeyPair::Public serverKey;
60 NetworkMessage serverKeyMessage;
61
62 if (clientSocket.waitForMessage(
63 serverKeyMessage, MessageProtocol::KEY_MESSAGE) != SOCKET_SUCCESS) {
64 std::cerr << "ERROR::Client.cpp: Server key message transfer failed."
65 << std::endl;
66 clientSocket.closeSocket();
67 return ConnectionStatus::CREDS_EXCHANGE_FAILED;
68 }
69
70 // If the received message was in any way erroneous, terminate the connection
71 if (serverKeyMessage.error()) {
72 std::cerr << "ERROR::Client.cpp: Server key message transfer failed."
73 << std::endl;
74 clientSocket.closeSocket();
75 return ConnectionStatus::CREDS_EXCHANGE_FAILED;
76 }
77
78 memcpy(&serverKey, serverKeyMessage.getMessageData(),
79 serverKeyMessage.getMessageSize());
80
81 // 3: Send random challenge to server encrypted under its public key
82 uint2048 challengeMessage;
83 uint64 *challenge = (uint64 *)&challengeMessage;
84 // We only want a 64 bit challenge, so leave the rest as 0s
85 CryptoSafeRandom::random(challenge, sizeof(uint64));
86 uint2048 encryptedChallenge = encrypt(challengeMessage, serverKey);
87 clientSocket.sendMessage(NetworkMessage(encryptedChallenge.rawData(),
88 sizeof(uint2048),
89 MessageProtocol::RSA_MESSAGE));
90
91 // 4: Receive signed challenge, nonce, session key and token, encrypted under
92 // client's public key
93 uint2048 signedEncryptedResponse;
94 NetworkMessage signedEncryptedResponseMessage;
95
96 if (clientSocket.waitForMessage(signedEncryptedResponseMessage,
97 MessageProtocol::RSA_MESSAGE) !=
98 SOCKET_SUCCESS) {
99 std::cerr << "ERROR::Client.cpp: Failed to retrieve signed response from "
100 "the server."
101 << std::endl;
102 clientSocket.closeSocket();
103 return ConnectionStatus::CREDS_EXCHANGE_FAILED;
104 }
105
106 // If the received message was in any way erroneous, terminate the connection
107 if (signedEncryptedResponseMessage.error()) {
108 std::cerr << "ERROR::Client.cpp: Failed to retrieve signed response from "
109 "the server."
110 << std::endl;
111 clientSocket.closeSocket();
112 return ConnectionStatus::CREDS_EXCHANGE_FAILED;
113 }
114
115 memcpy(&signedEncryptedResponse,
116 signedEncryptedResponseMessage.getMessageData(),
117 signedEncryptedResponseMessage.getMessageSize());
118
119 uint2048 responseValue = checkSignature(
120 decrypt(signedEncryptedResponse, clientKey.privateKey), serverSignature);
121 uint8 *responseData = (uint8 *)&responseValue;
122 uint64 responseChallenge;
123 uint32 nonce;
124
125 memcpy(&responseChallenge, responseData, sizeof(uint64));
126 memcpy(&nonce, responseData + sizeof(uint64), sizeof(uint32));
127 memcpy(&sessionKey, responseData + sizeof(uint32) + sizeof(uint64),
128 sizeof(AESKey));
129 memcpy(&sessionToken,
130 responseData + sizeof(uint32) + sizeof(uint64) + sizeof(AESKey),
131 sizeof(uint64));
132
133 if (responseChallenge != *challenge) {
134 clientSocket.closeSocket();
135 std::cerr << "ERROR::Client.cpp: Returned challenge from server incorrect. "
136 "Server failed to authenticate itself."
137 << std::endl;
138 return ConnectionStatus::CREDS_EXCHANGE_FAILED;
139 }
140
141 // 5: Client now has to gather and send a JWT to the server to authenticate
142 // themselves
143 QueryURL authUrl =
144 getMicrosoftAccountIDQueryURL(CLIENT_APPLICATION_ID, REDIRECT_URL, nonce);
145
146 authStringCallback(authUrl.url);
147
148 std::string authToken = openOneShotHTTPAuthenticationServer(
149 "127.0.0.1", 5000, "<html><body><h1>Thank you</h1></body></html>");
150
151 unsigned char *authMessageBuff =
152 (unsigned char *)alloca(sizeof(AuthMode) + authToken.size());
153
154 *((AuthMode *)authMessageBuff) = AuthMode::JWT;
155 memcpy(authMessageBuff + sizeof(AuthMode), authToken.c_str(),
156 authToken.size());
157
158 EncryptedNetworkMessage authMessage(
159 authMessageBuff, sizeof(AuthMode) + authToken.size(), sessionKey);
160 clientSocket.sendMessage(authMessage);
161
162 NetworkMessage authResponse;
163
164 if (clientSocket.receiveMessage(
165 authResponse, MessageProtocol::CONNECTION_RESPONSE_MESSAGE) ==
166 SOCKET_SUCCESS) {
167 if (!authResponse.error()) {
168 ConnectionResponse response =
169 *((ConnectionResponse *)authResponse.getMessageData());
170
171 switch (response) {
172 case ConnectionResponse::SUCCESS:
173 access = ClientAccess::LIMITED;
174 return ConnectionStatus::SUCCESS;
175 break;
176 case ConnectionResponse::SUCCESS_ADMIN:
177 access = ClientAccess::FULL;
178 return ConnectionStatus::SUCCESS;
179 case ConnectionResponse::FAILED:
180 return ConnectionStatus::INVALID_JWT;
181 break;
182 }
183 }
184 }
185
186 return ConnectionStatus::INVALID_JWT;
187}

333void Client::startClientLoop() {
334 // If we are already running, simply return
335 if (clientLoopRunning) {
336 return;
337 }
338
339 // Set the socket to non blocking
340 guardTCPSocketCode(clientSocket.setNonBlocking());
341
342 // Activate the loop thread - the loop will run for as long as this is true
343 clientLoopRunning = true;
344
345 // Create the thread pointing to the clientLoop function, called on this
346 // object
347 clientLoopThread = std::thread(&Client::clientLoop, this);
348}

350void Client::stopClientLoop() {
351 // If we aren't running, don't attempt to stop, simply return
352 if (!clientLoopRunning) {
353 return;
354 }
355
356 // Kill the loop - set this flag to false so the thread knows to exit
357 clientLoopRunning = false;
358
359 // Join and close the thread
360 clientLoopThread.join();
361}

391void Client::clientLoop() {
392 // Variables to hold the start and end times of each update frame (for
393 // calculating frame delta time)
394 std::chrono::time_point<std::chrono::system_clock> start, end;
395
396 while (clientLoopRunning) {
397 // Get the start time of this frame
398 start = std::chrono::system_clock::now();
399
400 sendQueueMutex.lock();
401 while (!sendQueue.empty()) {
402 NetworkMessage *message = sendQueue.front();
403 clientSocket.sendMessage(*message);
404 sendQueue.pop();
405 delete message;
406 }
407 sendQueueMutex.unlock();
408
409 EncryptedNetworkMessage rMessage;
410 if (clientSocket.receiveMessage(rMessage, MessageProtocol::AES_MESSAGE) ==
411 SOCKET_SUCCESS) {
412 if (!rMessage.error()) {
413 // decryptedMessage should not be owned by the client, and ownership
414 // should be passed to the response handler
415 void *&&decryptedMessage =
416 (void *)rMessage.decryptMessageData(sessionKey);
417
418 if (responseHandler) {
419 // Obviously moving a pointer is useless, but it indicates transfer of
420 // ownership, and that it is now the responsibility of the reciever to
421 // free.
422 responseHandler->onMessageReceived(std::move(decryptedMessage),
423 rMessage.getMessageSize());
424 }
425 }
426 }
427
428 // If the refresh rate is either negative or 0 it is invalid, so raise an
429 // error (otherwise there is the risk of division by 0 or running a while
430 // loop at maximum speed causing the process to overload the CPU!)
431 if (refreshRate <= 0) {
432 STD_ERROR("Refresh rate is <= 0. This is not allowed")
433 }
434
435 // Get the end time of this frame
436 end = std::chrono::system_clock::now();
437
438 // Calculate how long this frame took to execute
439 std::chrono::duration<double> frameElapsed = end - start;
440
441 // Calculate how much extra time must be waited to have the frame take the
442 // correct total time
443 double delta = (1.0 / refreshRate) - frameElapsed.count();
444
445 // If we have time to wait, wait it, otherwise don't (can't wait a negative
446 // time!)
447 if (delta > 0) {
448 // Sleep function in microseconds, so multiply by 10e6
449 // usleep(delta * 1e6);
450 std::this_thread::sleep_for(
451 std::chrono::microseconds((long long)(delta * 1e6)));
452 }
453 }
