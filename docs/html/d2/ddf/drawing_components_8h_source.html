1027template <typename T>
1028ComboboxComponentDataSource<T>::ComboboxComponentDataSource() {
1029 ComboboxDataSource::setAdapter(
1030 [](std::vector<unsigned>::const_iterator iter) {
1031 return DrawingComponentManager<T>::getComponentByHandle(*iter)
1032 .toDataElement();
1033 });
1034}
1038template <typename T>
1039void ComboboxComponentDataSource<T>::updateSource() {
1040 handleSet = DrawingComponentManager<T>::dataIndexSet();
1041 this->__begin = handleSet.begin();
1042 this->__end = handleSet.end();
1043
1044 DataSource::updateSource();
1045}

1050template <typename T>
1051void ComboboxComponentDataSource<T>::sort(
1052 const std::function<bool(const T &, const T &)> &comparator) {
1053 std::sort(
1054 handleSet.begin(), handleSet.end(), [comparator](unsigned a, unsigned b) {
1055 return comparator(DrawingComponentManager<T>::getComponentByHandle(a),
1056 DrawingComponentManager<T>::getComponentByHandle(b));
1057 });
1058}
1059
1063template <typename T>
1064void ComboboxComponentDataSource<T>::makeDistinct() {
1065 std::vector<unsigned>::const_iterator end = std::unique(
1066 handleSet.begin(), handleSet.end(), [this](unsigned a, unsigned b) {
1067 return DrawingComponentManager<T>::getComponentByHandle(a)
1068 .toDataElement(elementMode)
1069 .text == DrawingComponentManager<T>::getComponentByHandle(b)
1070 .toDataElement(elementMode)
1071 .text;
1072 });
1073 handleSet.erase(end, handleSet.end());
1074 this->__end = handleSet.end();
1075}
1076
1080template <typename T>
1081void ComboboxComponentDataSource<T>::setMode(unsigned mode) {
1082 ComboboxDataSource::setAdapter(
1083 [mode](std::vector<unsigned>::const_iterator iter) {
1084 return DrawingComponentManager<T>::getComponentByHandle(*iter)
1085 .toDataElement(mode);
1086 });
1087 elementMode = mode;
1088}

1093template <typename T>
1094bool ComboboxComponentDataSource<T>::empty() const {
1095 return handleSet.empty();
1096}
