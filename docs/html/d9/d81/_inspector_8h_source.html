191template<typename T>
192inline void Inspector::addComponentField(const QString& label, const std::function<void(const T&)>& fieldSetter,
193 ComboboxComponentDataSource<T>& source, unsigned defaultHandle) {
194 static_assert(std::is_base_of<DrawingComponent, T>::value, "Component Field type must inherit DrawingComponent.");
195
196 DynamicComboBox* fieldInput = new DynamicComboBox();
197 fieldInput->setDataSource(source);
198 fieldInput->setEditable(true);
199 fieldInput->setInsertPolicy(QComboBox::NoInsert);
200
201 if (defaultHandle != -1) {
202 fieldInput->setCurrentIndex(fieldInput->findData(defaultHandle));
203 }
204
205 QWidget::connect(fieldInput, qOverload<int>(&DynamicComboBox::currentIndexChanged), [this, fieldSetter, fieldInput](int index) {
206 if (index == -1) {
207 return;
208 }
209 fieldSetter(DrawingComponentManager<T>::getComponentByHandle(fieldInput->itemData(index).toInt()));
210 std::for_each(updateTriggers.begin(), updateTriggers.end(), [](const std::function<void()>& trigger) { trigger(); });
211 });
212
213 contents->addRow(label, fieldInput);
214};
