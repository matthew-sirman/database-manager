369// Creates all the source data for a given set of rows
370template<typename T> requires std::is_base_of_v<DatabaseRequestHandler::TableSourceData, T>
371void DatabaseRequestHandler::createSourceData(mysqlx::RowResult sourceRows) const {
372 // We statically assert that this is being used on a valid type. If there is an attempt to call this function on a type which does not derive
373 // from TableSourceData, there will be a compiler error.
374 static_assert(std::is_base_of<TableSourceData, T>::value, "Create Source Data can only be called with templates deriving TableSourceData.");
375
376 // Note: The calls to the other template methods within this method are all assumed to exist. If there is an attempt to call one which
377 // does not exist, a linker error will occur, meaning that the missing method must be implemented.
378
379 // Declare a list of the data elements for this type
380 std::vector<T> elements;
381 // Create a value for the buffer size. We will increment this with each element
382 // to find the total required buffer size.
383 unsigned bufferSize = sizeof(RequestType) + sizeof(unsigned);
384
385 // Initialise the handle value to 1. This will be incremented for each element. Handle 0 is reserved as a "null" option
386 // by the DrawingComponentManager.
387 unsigned handle = 1;
388
389 // We loop through each row in the source
390 // For each, we construct the data element(s). This is added to the elements list from inside this function.
391 constructDataElements<T>(sourceRows, handle, elements, bufferSize);
392
393 // Next we create the source data buffer with the size we have calculated. + size of extra networking paddings. Seems needed but not sure?
394 void *sourceBuffer = malloc(bufferSize + sizeof(uint64));
395
396 // We cast it to a byte buffer
397 unsigned char *buff = (unsigned char *)sourceBuffer;
398
399 // First we write the request type to the beginning of the buffer
400 *((RequestType *)buff) = getRequestType<T>();
401 buff += sizeof(RequestType);
402 // Next we write the number of elements, so the recipient knows how many elements to expect
403 *((unsigned *)buff) = elements.size();
404 buff += sizeof(unsigned);
405
406 // Then we loop over each element in the list
407 for (const T &element : elements) {
408 // We begin by writing the (program) handle to the buffer
409 *((unsigned *)buff) = element.handle;
410 buff += sizeof(unsigned);
411 // Then we write the component (database) ID to the buffer
412 *((unsigned *)buff) = element.id;
413 buff += sizeof(unsigned);
414
415 // Next we serialise the data element to the buffer and increment the buffer by the amount of
416 // space the element occupied.
417 serialiseDataElement<T>(element, &buff);
418 }
419
420 // Finally, once the buffer has been constructed, we send the buffer to the DrawingComponentManager of the correct type.
421 // This is where we use the T::ComponentType which will be the "true" type for each data type above. (e.g. ProductData -> Product)
422 DrawingComponentManager<typename T::ComponentType>::sourceComponentTable(std::move(sourceBuffer), bufferSize + 4);
423}
