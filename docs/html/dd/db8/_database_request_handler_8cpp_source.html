16void DatabaseRequestHandler::onMessageReceived(Server &caller,
17 const ClientHandle &clientHandle,
18 void *&&message,
19 unsigned int messageSize) {
20 Logger::log("Message Type: " +
21 std::to_string(static_cast<int>(getDeserialiseType(message))));
22 switch (getDeserialiseType(message)) {
23 case RequestType::REPEAT_TOKEN_REQUEST:
24 caller.sendRepeatToken(clientHandle,
25 (unsigned)RequestType::REPEAT_TOKEN_REQUEST);
26 break;
27 case RequestType::USER_EMAIL_REQUEST:
28 caller.sendEmailAddress(clientHandle,
29 (unsigned)RequestType::USER_EMAIL_REQUEST);
30 break;
31 case RequestType::DRAWING_SEARCH_QUERY: {
32 DatabaseSearchQuery &query =
33 DatabaseSearchQuery::deserialise(std::move(message));
34 std::vector<DrawingSummary> summaries =
35 caller.databaseManager().executeSearchQuery(query);
36 delete &query;
37
38 DrawingSummaryCompressionSchema summaryCompressionSchema =
39 compressionSchema(&caller.databaseManager());
40
41 unsigned char *responseBuffer = (unsigned char *)alloca(
42 sizeof(RequestType) + sizeof(DrawingSummaryCompressionSchema) +
43 sizeof(unsigned) +
44 summaries.size() * summaryCompressionSchema.maxCompressedSize());
45
46 unsigned index = 0;
47
48 *((RequestType *)responseBuffer) = RequestType::DRAWING_SEARCH_QUERY;
49 index += sizeof(RequestType);
50
51 memcpy(responseBuffer + index, &summaryCompressionSchema,
52 sizeof(DrawingSummaryCompressionSchema));
53 index += sizeof(DrawingSummaryCompressionSchema);
54
55 *((unsigned *)(responseBuffer + index)) = summaries.size();
56 index += sizeof(unsigned);
57
58 for (const DrawingSummary &summary : summaries) {
59 summaryCompressionSchema.compressSummary(summary,
60 responseBuffer + index);
61 index += summaryCompressionSchema.compressedSize(summary);
62 }
63
64 caller.addMessageToSendQueue(clientHandle, responseBuffer, index);
65
66 break;
67 }
68 case RequestType::DRAWING_INSERT: {
69 DrawingInsert &drawingInsert =
70 DrawingInsert::deserialise(std::move(message));
71
72 if (drawingInsert.drawingData.has_value()) {
73 DrawingInsert response;
74
75 response.responseEchoCode = drawingInsert.responseEchoCode;
76
77 switch (caller.databaseManager().drawingExists(
78 drawingInsert.drawingData->drawingNumber())) {
79 case DatabaseManager::DrawingExistsResponse::EXISTS:
80 if (drawingInsert.forcing()) {
81 if (caller.databaseManager().insertDrawing(drawingInsert)) {
82 response.insertResponseCode = DrawingInsert::SUCCESS;
83 } else {
84 response.insertResponseCode = DrawingInsert::FAILED;
85 }
86 } else {
87 response.insertResponseCode = DrawingInsert::DRAWING_EXISTS;
88 }
89 break;
90 case DatabaseManager::DrawingExistsResponse::NOT_EXISTS:
91 if (caller.databaseManager().insertDrawing(drawingInsert)) {
92 response.insertResponseCode = DrawingInsert::SUCCESS;
93 } else {
94 response.insertResponseCode = DrawingInsert::FAILED;
95 }
96 break;
97 case DatabaseManager::DrawingExistsResponse::R_ERROR:
98 response.insertResponseCode = DrawingInsert::FAILED;
99 break;
100 }
101
102 if (response.insertResponseCode == DrawingInsert::SUCCESS) {
103 setCompressionSchemaDirty();
104 caller.changelogMessage(
105 clientHandle,
106 "Added drawing " + drawingInsert.drawingData->drawingNumber());
107 }
108
109 delete &drawingInsert;
110
111 unsigned responseSize = response.serialisedSize();
112 void *responseBuffer = alloca(responseSize);
113 response.serialise(responseBuffer);
114
115 caller.addMessageToSendQueue(clientHandle, responseBuffer,
116 responseSize);
117
118 NextDrawing automatic, manual;
119 automatic.drawingType = NextDrawing::DrawingType::AUTOMATIC;
120 manual.drawingType = NextDrawing::DrawingType::MANUAL;
121
122 automatic.drawingNumber =
123 caller.databaseManager().nextAutomaticDrawingNumber();
124 manual.drawingNumber =
125 caller.databaseManager().nextManualDrawingNumber();
126
127 unsigned autoBufferSize = automatic.serialisedSize();
128 void *autoBuffer = alloca(autoBufferSize);
129 automatic.serialise(autoBuffer);
130
131 unsigned manualBufferSize = manual.serialisedSize();
132 void *manualBuffer = alloca(manualBufferSize);
133 manual.serialise(manualBuffer);
134 caller.broadcastMessage(autoBuffer, autoBufferSize);
135 caller.broadcastMessage(manualBuffer, manualBufferSize);
136 }
137
138 break;
139 }

143 caller.databaseManager().sourceTable("products"));

565DrawingSummaryCompressionSchema DatabaseRequestHandler::compressionSchema(
566 DatabaseManager *dbManager) {
567 if (schemaDirty) {
568 if (!dbManager) {
569 STD_ERROR("Database manager not set up. No connection to database.");
570 }
571
572 if (DrawingComponentManager<Aperture>::dirty()) {
573 if (DrawingComponentManager<ApertureShape>::dirty()) {
574 createSourceData<ApertureShapeData>(
575 dbManager->sourceTable("aperture_shapes"));
576 }
577
578 createSourceData<ApertureData>(dbManager->sourceTable("apertures"));
579 }
580
581 if (DrawingComponentManager<Material>::dirty()) {
582 createSourceData<MaterialData>(dbManager->sourceMultipleTable(
583 "material_prices", "materials", "material_id"));
584 }
585
586 if (DrawingComponentManager<SideIronPrice>::dirty()) {
587 createSourceData<SideIronPriceData>(
588 dbManager->sourceTable("side_iron_prices"));
589 }
590
591 unsigned maxMatID, maxThicknessHandle, maxApertureHandle;
592 float maxWidth, maxLength, maxLapSize;
593 unsigned char maxDrawingLength;
594 unsigned char maxBarSpacingCount;
595 float maxBarSpacing;
596 unsigned char maxExtraApertureCount;
597
598 dbManager->getCompressionSchemaDetails(
599 maxMatID, maxWidth, maxLength, maxLapSize, maxBarSpacingCount,
600 maxBarSpacing, maxDrawingLength, maxExtraApertureCount);
601 maxThicknessHandle = DrawingComponentManager<Material>::maximumHandle();
602 maxApertureHandle = DrawingComponentManager<Aperture>::maximumHandle();
603
604 schema = DrawingSummaryCompressionSchema(
605 maxMatID, maxWidth, maxLength, maxThicknessHandle, maxLapSize,
606 maxApertureHandle, maxBarSpacingCount, maxBarSpacing, maxDrawingLength,
607 maxExtraApertureCount);
608
609 schemaDirty = false;
610 }
611
612 return schema;
613}

629template <>
630void DatabaseRequestHandler::constructDataElements(
631 mysqlx::RowResult &productRow, unsigned &handle,
632 std::vector<DatabaseRequestHandler::ProductData> &elements,
633 unsigned &sizeValue) const {
634 for (mysqlx::Row row : productRow.fetchAll()) {
635 if (!row.isNull()) {
636 ProductData data;
637 data.handle = handle++;
638 data.id = row.get(0).get<unsigned int>();
639 data.name = row.get(1).get<std::string>();
640 sizeValue +=
641 sizeof(unsigned) * 2 + sizeof(unsigned char) + data.name.size();
642
643 elements.push_back(data);
644 }
645 }
646}

682template <>
683void DatabaseRequestHandler::constructDataElements(
684 mysqlx::RowResult &apertureRow, unsigned &handle,
685 std::vector<DatabaseRequestHandler::ApertureData> &elements,
686 unsigned &sizeValue) const {
687 for (mysqlx::Row row : apertureRow.fetchAll()) {
688 if (row[6].isNull()) {
689 ERROR_RAW_SAFE("Aperture with missing shape ID detected.", std::cerr);
690 return;
691 }
692 ApertureData data;
693 data.handle = handle++;
694 data.id = row[0];
695 data.width = row[1].isNull() ? 0 : row[1].get<float>();
696 data.length = row[2].isNull() ? 0 : row[2].get<float>();
697 data.baseWidth = row[3].isNull() ? 0 : row[3].get<unsigned>();
698 data.baseLength = row[4].isNull() ? 0 : row[4].get<unsigned>();
699 data.quantity = row[5].isNull() ? 0 : row[5].get<unsigned>();
700 data.shapeID =
701 DrawingComponentManager<ApertureShape>::findComponentByID(row[6])
702 .handle();
703 data.isNibble = row[7].get<bool>();
704 if (data.isNibble) {
705 data.nibbleApertureId = row[8].get<unsigned>();
706 }
707
708 sizeValue += sizeof(unsigned) * 2 + sizeof(float) * 2 +
709 sizeof(unsigned short) * 3 + sizeof(unsigned) + sizeof(bool) +
710 (data.isNibble ? sizeof(unsigned) : 0);
711
712 elements.push_back(data);
713 }
714}

722template <>
723void DatabaseRequestHandler::serialiseDataElement(
724 const DatabaseRequestHandler::ApertureData &element,
725 unsigned char **buffer) const {
726 *((float *)*buffer) = element.width;
727 *buffer += sizeof(float);
728 *((float *)*buffer) = element.length;
729 *buffer += sizeof(float);
730 *((unsigned short *)*buffer) = element.baseWidth;
731 *buffer += sizeof(unsigned short);
732 *((unsigned short *)*buffer) = element.baseLength;
733 *buffer += sizeof(unsigned short);
734 *((unsigned *)*buffer) = element.shapeID;
735 *buffer += sizeof(unsigned);
736 *((unsigned short *)*buffer) = element.quantity;
737 *buffer += sizeof(unsigned short);
738 *(*buffer)++ = element.isNibble;
739 if (element.isNibble) {
740 *((unsigned *)*buffer) = element.nibbleApertureId;
741 *buffer += sizeof(unsigned);
742 }
743}
